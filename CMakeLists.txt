cmake_minimum_required(VERSION 3.21)
project(modern-standby-auditor VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "modern-standby-auditor requires Windows")
endif()
option(MSA_ANALYZE "Run MSVC code analysis" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(msa_core src/audit.cpp)
target_include_directories(msa_core PUBLIC src)
target_compile_definitions(msa_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000000)
target_link_libraries(msa_core PUBLIC PowrProf Advapi32)
add_executable(modern-standby-auditor src/main.cpp)
target_link_libraries(modern-standby-auditor PRIVATE msa_core)
if(MSVC)
  target_sources(modern-standby-auditor PRIVATE src/app.manifest)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(msa_tests tests/tests.cpp)
  target_link_libraries(msa_tests PRIVATE msa_core)
  add_test(NAME unit COMMAND msa_tests)
  add_test(NAME help COMMAND modern-standby-auditor --help)
  set_tests_properties(help PROPERTIES PASS_REGULAR_EXPRESSION "Usage:")
  add_test(NAME version COMMAND modern-standby-auditor --version)
  set_tests_properties(version PROPERTIES PASS_REGULAR_EXPRESSION "1.0.0")
  add_test(NAME json COMMAND modern-standby-auditor --json)
  set_tests_properties(json PROPERTIES PASS_REGULAR_EXPRESSION "schema_version")
  add_test(NAME malformed COMMAND modern-standby-auditor --invalid)
  set_tests_properties(malformed PROPERTIES WILL_FAIL TRUE)
  add_test(NAME conflicting COMMAND modern-standby-auditor --json --help)
  set_tests_properties(conflicting PROPERTIES WILL_FAIL TRUE)
  add_test(NAME safety COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_SOURCE_DIR} -P ${CMAKE_SOURCE_DIR}/tests/safety.cmake)
endif()
foreach(target msa_core modern-standby-auditor msa_tests)
  if(TARGET ${target} AND MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    if(MSA_ANALYZE)
      target_compile_options(${target} PRIVATE /analyze)
    endif()
  endif()
endforeach()
install(TARGETS modern-standby-auditor RUNTIME DESTINATION .)
install(FILES README.md LICENSE VALIDATION.md schema.json DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "modern-standby-auditor-${PROJECT_VERSION}-windows-${CMAKE_SYSTEM_PROCESSOR}")
include(CPack)
